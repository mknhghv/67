local ActiveTween = nil

local LastNoclipApply = 0

local function ApplyNoclip(character)
    if not character then return end
    for _, v in ipairs(character:GetChildren()) do
        if v:IsA("BasePart") and v.CanCollide then
            v.CanCollide = false
        end
    end
end

local function forceNoclip()
    local now = tick()
    if now - LastNoclipApply < 0.2 then
        return
    end
    LastNoclipApply = now
    ApplyNoclip(LocalPlayer.Character)
end

local function EnsureNoclipLoop()
    if NoclipConnection then return end
    NoclipConnection = RunService.Heartbeat:Connect(function()
        if not ActiveTween or ActiveTween.PlaybackState ~= Enum.PlaybackState.Playing then
            if NoclipConnection then
                NoclipConnection:Disconnect()
                NoclipConnection = nil
            end
            return
        end
        ApplyNoclip(LocalPlayer.Character)
    end)
end

local function topos(Pos)
    if not LocalPlayer or not LocalPlayer.Character then return end

    local HRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local Humanoid = LocalPlayer.Character:FindFirstChild("Humanoid")
    if not HRP or not Humanoid then return end

    if ActiveTween
    and ActiveTween.PlaybackState == Enum.PlaybackState.Playing
    and ActiveTweenTarget
    and (ActiveTweenTarget.Position - Pos.Position).Magnitude <= 4 then
        return
    end

    HRP.AssemblyLinearVelocity = ZERO
    HRP.AssemblyAngularVelocity = ZERO

    local Distance = (Pos.Position - HRP.Position).Magnitude
    if Distance <= 5 then
        if ActiveTween then
            ActiveTween:Cancel()
            ActiveTween = nil
            ActiveTweenTarget = nil
        end
        HRP.CFrame = Pos
        return
    end

    local Speed = 350
    if Distance < 150 then
        Speed = 600
    elseif Distance < 250 then
        Speed = 450
    elseif Distance < 500 then
        Speed = 400
    elseif Distance < 750 then
        Speed = 300
    end

    if ActiveTween then
        ActiveTween:Cancel()
        ActiveTween = nil
    end

    ActiveTweenTarget = Pos
    ActiveTween = TweenService:Create(
        HRP,
        TweenInfo.new(Distance / Speed, Enum.EasingStyle.Linear),
        {CFrame = Pos}
    )

    EnsureNoclipLoop()
    ActiveTween:Play()
end

local function StopTween()
    if ActiveTween then
        ActiveTween:Cancel()
        ActiveTween = nil
    end
    ActiveTweenTarget = nil

    if NoclipConnection then
        NoclipConnection:Disconnect()
        NoclipConnection = nil
    end

    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local bodyClip = LocalPlayer.Character.HumanoidRootPart:FindFirstChild("BodyClip")
        if bodyClip then bodyClip:Destroy() end
    end
end

local function isShadow(name)
    local lower = string.lower(name)
    return string.find(lower, "shadow") ~= nil
        or string.find(name, "'s Shadow") ~= nil
        or string.find(name, "\u{2019}s Shadow") ~= nil
end

-- ====================================================================
-- 【高階快速攻擊主邏輯（全武器/空手開放版）】
-- ====================================================================
task.spawn(function()
    while true do
        if Config.FastAttackDelay > 0 then
            task.wait(Config.FastAttackDelay)
        else
            local delta = tick() - lastAttackTime
            if delta < 0.015 then
                task.wait(0.015 - delta)
            else
                task.wait()
            end
        end

        if not _G.FastAttack then continue end

        local Character = player.Character
        local hrp = Character and Character:FindFirstChild('HumanoidRootPart')
        if not Character or not hrp then continue end

        local hum = Character:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then continue end

        local attackRange = Config.AttackRange or 500

        local BladeHits = {}
        local MobFolders = { workspace:FindFirstChild("Enemies"), workspace:FindFirstChild("Characters") }
        for _, Mobs in ipairs(MobFolders) do
            if not Mobs then continue end
            if Mobs.Name == "Enemies" and not Config.FastAttackMob then continue end
            if Mobs.Name == "Characters" and not Config.FastAttackPlayer then continue end
            for _, Mob in ipairs(Mobs:GetChildren()) do
                if Mob == Character or not Mob:IsA("Model") or isShadow(Mob.Name) or Mob.Name == "Blank Buddy" then continue end
                local mobHrp = Mob:FindFirstChild('HumanoidRootPart')
                local mobHum = Mob:FindFirstChild('Humanoid')
                if mobHrp and mobHum and mobHum.Health > Config.HealthMobThreshold then
                    if (mobHrp.Position - hrp.Position).Magnitude <= attackRange then
                        BladeHits[#BladeHits + 1] = {Mob, mobHrp}
                    end
                end
            end
        end

        if #BladeHits > 0 then
            local targetMob = BladeHits[1][1]
            local _Head = targetMob:FindFirstChild('Head') or BladeHits[1][2]
            if _Head then
                lastAttackTime = tick()
                local sessionToken = generateRegisterHitToken()

                if MobDamageTracker[targetMob] then
                    MobDamageTracker[targetMob].lastHit = tick()
                end

                -- 直接觸發打擊封包
                RegisterAttack:FireServer(0.01)
                RegisterHit:FireServer(_Head, BladeHits, {}, sessionToken)

                if Config.RemoveAnimationFast and CancelEquipAnim then
                    CancelEquipAnim:FireServer()
                end

                if Config.DoubleAttack then
                    RegisterAttack:FireServer(0.01)
                    RegisterHit:FireServer(_Head, BladeHits, {}, sessionToken)
                    if Config.RemoveAnimationFast and CancelEquipAnim then
                        CancelEquipAnim:FireServer()
                    end
                end

                if Config.BypassFastAttack and remoteEvent then
                    task.spawn(function()
                        pcall(function()
                            if NetModuleRequire then
                                NetModuleRequire:RemoteEvent('RegisterHit', true)
                            end
                            remoteEvent:FireServer(
                                string.gsub('RE/RegisterHit', '.', function(p)
                                    return string.char(bit32.bxor(string.byte(p), math.floor(workspace:GetServerTimeNow() / 10 % 10) + 1))
                                end),
                                bit32.bxor(remoteId + 909090, ReplicatedStorage.Modules.Net.seed:InvokeServer() * 2),
                                _Head, BladeHits
                            )
                        end)
                    end)
                end
            end
        end
    end
end)

local function IsFarmTargetMob(name)
    if _G.FarmLocation == "左邊" or _G.FarmLocation == "左边" then
        return LeftMobs[name] == true
    elseif _G.FarmLocation == "右邊" or _G.FarmLocation == "右边" then
        return RightMobs[name] == true
    elseif _G.FarmLocation == "左右都打" then
        return LeftMobs[name] == true or RightMobs[name] == true
    end
    return false
end

local LeftOnlyPoints = { leftSpawnCFrame }
local RightOnlyPoints = { rightSpawnCFrame }
local BothPoints = { leftSpawnCFrame, rightSpawnCFrame }

local function GetSpawnPoints()
    if _G.FarmLocation == "右邊" or _G.FarmLocation == "右边" then
        return RightOnlyPoints
    elseif _G.FarmLocation == "左右都打" then
        return BothPoints
    end
    return LeftOnlyPoints
end

local function ResetFarmWaitIndex()
    FarmWaitIndex = 1
    FarmSpotDwellUntil = 0
end

local function GetCurrentWaitPos()
    local points = GetSpawnPoints()
    if #points == 0 then return leftSpawnCFrame end
    if FarmWaitIndex < 1 or FarmWaitIndex > #points then
        FarmWaitIndex = 1
    end
    return points[FarmWaitIndex]
end

local function AdvanceWaitPos(hrp)
    local points = GetSpawnPoints()
    if #points <= 1 then
        return GetCurrentWaitPos()
    end

    local current = GetCurrentWaitPos()
    if not current then return leftSpawnCFrame end

    if hrp and (hrp.Position - current.Position).Magnitude <= 15 then
        local now = tick()
        if FarmSpotDwellUntil <= 0 then
            FarmSpotDwellUntil = now + FarmSpotDwellTime
        elseif now >= FarmSpotDwellUntil then
            FarmWaitIndex = FarmWaitIndex + 1
            if FarmWaitIndex > #points then
                FarmWaitIndex = 1
            end
            FarmSpotDwellUntil = 0
        end
    else
        if FarmSpotDwellUntil > 0 and hrp and (hrp.Position - current.Position).Magnitude > 80 then
            FarmSpotDwellUntil = 0
        end
    end

    return GetCurrentWaitPos()
end

local CachedFarmTarget = nil
local LastKillTime = 0

local function IsAliveMob(mob)
    if not mob or not mob.Parent then return false end
    local root = mob:FindFirstChild("HumanoidRootPart")
    local hum = mob:FindFirstChild("Humanoid")
    return root ~= nil and hum ~= nil and hum.Health > 0 and mob.PrimaryPart ~= nil
end

local function GetFarmTarget()
    local now = tick()
    
    if CachedFarmTarget then
        if IsAliveMob(CachedFarmTarget) then
            return CachedFarmTarget
        else
            CachedFarmTarget = nil
            LastKillTime = now
        end
    end

    local character = LocalPlayer.Character
    if not character then return nil end
    local myRoot = character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local myPos = myRoot.Position

    local enemiesFolder = workspace:FindFirstChild("Enemies")
    if not enemiesFolder then return nil end

    local nearestMob = nil
    local minDistance = math.huge
    local children = enemiesFolder:GetChildren()

    for i = 1, #children do
        local mob = children[i]
        local name = mob.Name
        if name == "Tyrant of the Skies" and IsAliveMob(mob) then
            CachedFarmTarget = mob
            return mob
        end

        if IsFarmTargetMob(name) and IsAliveMob(mob) then
            local root = mob:FindFirstChild("HumanoidRootPart")
            local dist = (root.Position - myPos).Magnitude
            if _G.FarmLocation == "左右都打" then
                if dist < minDistance then
                    minDistance = dist
                    nearestMob = mob
                end
            else
                if dist < minDistance and dist < 3000 then
                    minDistance = dist
                    nearestMob = mob
                end
            end
        end
    end

    CachedFarmTarget = nearestMob
    return nearestMob
end

local bringMobsConnection = nil

local function StartBringMobs()
    if bringMobsConnection then return end
    
    bringMobsConnection = RunService.Heartbeat:Connect(function()
        if not _G.G_BringMobs then return end
        
        pcall(function()
            local char = LocalPlayer.Character
            if not char then return end
            
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            
            local hum = char:FindFirstChild("Humanoid")
            if not hum or hum.Health <= 0 then return end
            
            if sethiddenproperty then
                sethiddenproperty(LocalPlayer, "SimulationRadius", math.huge)
            end
            
            local targetPos = hrp.Position
            local enemies = Workspace:FindFirstChild("Enemies")
            if not enemies then return end
            
            local enemyList = enemies:GetChildren()
            
            for i = 1, #enemyList do
                local enemy = enemyList[i]
                if enemy:IsA("Model") then
                    local enemyHum = enemy:FindFirstChild("Humanoid")
                    local enemyRoot = enemy:FindFirstChild("HumanoidRootPart")
                    
                    if enemyHum and enemyRoot and enemyHum.Health > 0 then
                        local distance = (enemyRoot.Position - targetPos).Magnitude
                        
                        if distance <= 3000 then
                            local bv = enemyRoot:FindFirstChild("BodyVelocity")
                            if not bv then
                                bv = Instance.new("BodyVelocity")
                                bv.Name = "BodyVelocity"
                                bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
                                bv.Velocity = Vector3.zero
                                bv.Parent = enemyRoot
                            end
                            
                            if distance > 18 then
                                local isOwner = false
                                if isnetworkowner then
                                    isOwner = isnetworkowner(enemyRoot)
                                else
                                    isOwner = enemyRoot.ReceiveAge == 0 and not enemyRoot.Anchored
                                end
                                
                                if isOwner then
                                    enemyRoot.CFrame = CFrame.new(targetPos.X, targetPos.Y - 18, targetPos.Z)
                                end
                            end
                            
                            enemyRoot.CanCollide = false
                            enemyHum.WalkSpeed = 0
                            enemyHum.JumpPower = 0
                        end
                    end
                end
            end
        end)
    end)
end

local CachedIslandModel = nil
local CachedArena = nil
local CachedEyes = nil
local NextEyeCacheRefresh = 0
local NextTreeCacheRefresh = 0
local CachedTreeList = nil
local NextEagleEyeCheck = 0
local CachedEagleEyeResult = false

local function GetIslandModel()
    if CachedIslandModel and CachedIslandModel.Parent then return CachedIslandModel end
    local map = workspace:FindFirstChild("Map")
    local tiki = map and map:FindFirstChild("TikiOutpost")
    CachedIslandModel = tiki and tiki:FindFirstChild("IslandModel") or nil
    return CachedIslandModel
end

local function RefreshEyeCache(islandModel)
    local eyes = {}
    local descendants = islandModel:GetDescendants()
    for i = 1, #descendants do
        local v = descendants[i]
        local name = v.Name
        if name == "Eye1" or name == "Eye2" or name == "Eye3" or name == "Eye4" then
            eyes[name] = v
        end
    end
    CachedEyes = eyes
    NextEyeCacheRefresh = tick() + 5
end

local function checkEagleEye()
    local now = tick()
    if now < NextEagleEyeCheck then return CachedEagleEyeResult end
    NextEagleEyeCheck = now + 1

    local islandModel = GetIslandModel()
    if not islandModel then return false end

    if not CachedEyes or now >= NextEyeCacheRefresh then RefreshEyeCache(islandModel) end

    local eyes = CachedEyes
    local e1, e2, e3, e4 = eyes.Eye1, eyes.Eye2, eyes.Eye3, eyes.Eye4
    if not (e1 and e1.Parent and e2 and e2.Parent and e3 and e3.Parent and e4 and e4.Parent) then
        RefreshEyeCache(islandModel)
        eyes = CachedEyes
        e1, e2, e3, e4 = eyes.Eye1, eyes.Eye2, eyes.Eye3, eyes.Eye4
    end

    if not (e1 and e2 and e3 and e4) then return false end

    CachedEagleEyeResult = (tonumber(e1.Transparency) == 0)
        and (tonumber(e2.Transparency) == 0)
        and (tonumber(e3.Transparency) == 0)
        and (tonumber(e4.Transparency) == 0)
    return CachedEagleEyeResult
end

local function GetArena()
    if CachedArena and CachedArena.Parent then return CachedArena end
    local islandModel = GetIslandModel()
    if not islandModel then return nil end
    local descendants = islandModel:GetDescendants()
    for i = 1, #descendants do
        local v = descendants[i]
        if v.Name == "EagleBossArena" then
            CachedArena = v
            return CachedArena
        end
    end
    return nil
end

local function RefreshTreeCache(arena)
    local trees = {}
    local descendants = arena:GetDescendants()
    for i = 1, #descendants do
        local obj = descendants[i]
        if string.lower(obj.Name) == "tree" then
            if (obj:IsA("Model") and obj.PrimaryPart) or obj:IsA("BasePart") then
                trees[#trees + 1] = obj
            end
        end
    end
    CachedTreeList = trees
    NextTreeCacheRefresh = tick() + 3
end

local function getNearestTree()
    local character = LocalPlayer.Character
    if not character then return nil end
    local myRoot = character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local myPos = myRoot.Position

    local arena = GetArena()
    if not arena then return nil end

    local now = tick()
    if not CachedTreeList or now >= NextTreeCacheRefresh then RefreshTreeCache(arena) end

    local nearestTree = nil
    local minDistance = math.huge
    local trees = CachedTreeList
    for i = 1, #trees do
        local obj = trees[i]
        if obj and obj.Parent then
            local pos = nil
            if obj:IsA("Model") and obj.PrimaryPart then
                pos = obj.PrimaryPart.Position
            elseif obj:IsA("BasePart") then
                pos = obj.Position
            end
            if pos then
                local dist = (pos - myPos).Magnitude
                if dist < minDistance then
                    minDistance = dist
                    nearestTree = obj
                end
            end
        end
    end
    return nearestTree
end

local function fireSkullGuitar(targetPos)
    local character = LocalPlayer.Character
    if not character then return end

    local humanoid = character:FindFirstChild("Humanoid")
    if not humanoid then return end

    -- 自動裝備吉他
    local guitar = character:FindFirstChild("Skull Guitar")

        local backpack = LocalPlayer:FindFirstChild("Backpack")
        if backpack and backpack:FindFirstChild("Skull Guitar") then
            humanoid:EquipTool(backpack["Skull Guitar"])
            task.wait(0.1)
        end
    end

    guitar = character:FindFirstChild("Skull Guitar")
    if not guitar then return end

    -- 朝下攻擊的位置計算（Y軸向下降 15 單位）
    local downPos = Vector3.new(targetPos.X, targetPos.Y - 15, targetPos.Z)

    -- 搜尋 Tool 內的 RemoteFunction/RemoteEvent 進行 InvokeServer
    local remote = guitar:FindFirstChildOfClass("RemoteFunction") or guitar:FindFirstChildOfClass("RemoteEvent") or humanoid:FindFirstChildOfClass("RemoteFunction")
    
    pcall(function()
        if remote and remote:IsA("RemoteFunction") then
            remote:InvokeServer("TAP", downPos)
        elseif remote and remote:IsA("RemoteEvent") then
            remote:FireServer("TAP", downPos)
        end
    end)
end

                    if not myRoot or not hum then return end

                    myRoot.AssemblyLinearVelocity = ZERO
                    myRoot.AssemblyAngularVelocity = ZERO

                    if _G.isHuntingBird then
                        BringMob = false
                        BringCount = 0
                        return
                    end

                    local targetMob = GetFarmTarget()
                    if targetMob and IsAliveMob(targetMob) then
                        local targetRoot = targetMob:FindFirstChild("HumanoidRootPart")
                        if not targetRoot then return end

                        forceNoclip()
                        hasReachedSpawn = false

                        BringMob = true
                        MonFarm = targetMob.Name
                        FarmPos = targetRoot.CFrame
                        
                        targetRoot.CanCollide = false
                        local head = targetMob:FindFirstChild("Head")
                        if head then head.CanCollide = false end

                        local targetCFrame = targetRoot.CFrame * CFrame.new(0, 12, 0)
                        local dist = (myRoot.Position - targetCFrame.Position).Magnitude
                        if dist > 6 then
                            topos(targetCFrame)
                        else
                            StopTween()
                            myRoot.CFrame = targetCFrame
                        end
                    else
                        BringMob = false
                        BringCount = 0
                        FarmPos = nil
                        forceNoclip()

                        if now - LastKillTime < 0.35 then
                            StopTween()
                            return
                        end

                        local waitPos = AdvanceWaitPos(myRoot)
                        local dist = (myRoot.Position - waitPos.Position).Magnitude
                        if dist > 6 then
                            topos(waitPos)
                        else
                            StopTween()
                            myRoot.CFrame = waitPos
                            hasReachedSpawn = true
                        end
                    end
                end)
            end
        else
            if FarmConnection then
                FarmConnection:Disconnect()
                FarmConnection = nil
                hasReachedSpawn = false
                ResetFarmWaitIndex()
            end
            BringMob = false
            BringCount = 0
            FarmPos = nil
            CachedFarmTarget = nil
            StopTween()
        end
    end
end)

local function GetWeapon()
    if not LocalPlayer then return nil end

    local targetType = _G.WeaponType or "Melee"
    local character = LocalPlayer.Character
    if character then
        for _, item in ipairs(character:GetChildren()) do
            if item:IsA("Tool") and item.ToolTip == targetType then return item end
        end
    end

    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and item.ToolTip == targetType then return item end
        end
    end
    return nil
end

task.spawn(function()
    while task.wait(0.5) do
        if _G.AutoEquipWeapon and _G.HuntingType ~= "Tree" then
            pcall(function()
                local character = LocalPlayer.Character
                local humanoid = character and character:FindFirstChild("Humanoid")
                if not character or not humanoid or humanoid.Health <= 0 then return end

                local weapon = GetWeapon()
                if weapon and weapon.Parent ~= character then humanoid:EquipTool(weapon) end
            end)
        end
    end
end)
